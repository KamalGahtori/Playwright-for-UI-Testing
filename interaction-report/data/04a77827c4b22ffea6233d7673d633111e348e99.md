  1   | // tests/interaction/interaction.spec.js
  2   | // ─────────────────────────────────────────────────────────────────────
  3   | // Interaction Audit
  4   | //
  5   | // One test per endpoint. Each test:
  6   | //   1. Navigates to the page.
  7   | //   2. preparePage — full 3-pass load strategy (forces all lazy assets,
  8   | //                    waits for layout stability). No stabilizePage —
  9   | //                    overlays and widgets are intentionally left in place
  10  | //                    so they are checked as interactive elements.
  11  | //   3. runInteractionAudit — dynamic element scan + accessibility checks:
  12  | //        buttons visible & enabled | links reachable (HEAD) |
  13  | //        inputs visible & enabled (no text entry anywhere) |
  14  | //        selects enabled | nav dropdowns hover-accessible |
  15  | //        carousel nav buttons visible | accordions enabled
  16  | //
  17  | // WHAT IS CHECKED:
  18  | //   Every interactive element discovered at runtime. Zero hardcoded selectors.
  19  | //
  20  | // WHAT IS SKIPPED (never interacted with):
  21  | //   Email fields, captcha inputs, file inputs, hidden/password inputs.
  22  | //   No text is entered anywhere on any page.
  23  | //
  24  | // FAILURE POLICY:
  25  | //   Hard fail on: disabled buttons, inaccessible inputs, internal link errors (4xx/5xx).
  26  | //   Warn on: external link 4xx (server may block HEAD), carousel nav not visible.
  27  | // ─────────────────────────────────────────────────────────────────────
  28  | 
  29  | const { test, expect }       = require('../../utils/base-fixtures');
  30  | const endpoints               = require('../../endpoints.config');
  31  | const { runInteractionAudit } = require('../../utils/interaction-engine');
  32  | 
  33  | test.describe('Interaction Audit', () => {
  34  | 
  35  |   for (const endpoint of endpoints) {
  36  | 
  37  |     test(`[${endpoint.group}][${endpoint.id}] Interaction Audit`, async ({ page, preparePage }) => {
  38  | 
  39  |       const projectName = test.info().project.name;
  40  | 
  41  |       // ── Navigate ───────────────────────────────────────────────────
  42  |       await test.step(`Navigating to ${endpoint.path}`, async () => {
  43  |         try {
  44  |           const response = await page.goto(endpoint.path);
  45  |           if (!response || response.status() >= 400) {
  46  |             throw new Error(
  47  |               `\n\n⚠️ NAVIGATION FAILURE ⚠️\n` +
  48  |               `Could not reach: ${endpoint.path} (HTTP ${response?.status() ?? 'N/A'})\n`
  49  |             );
  50  |           }
  51  |         } catch (err) {
  52  |           if (err.message.includes('NAVIGATION FAILURE')) throw err;
  53  |           throw new Error(
  54  |             `\n\n⚠️ CONNECTION ERROR ⚠️\n` +
  55  |             `Failed to connect to: ${endpoint.path}\n` +
  56  |             `Error: ${err.message}\n`
  57  |           );
  58  |         }
  59  |       });
  60  | 
  61  |       // ── Phase 1: Full page warm-up ─────────────────────────────────
  62  |       // Force-loads all lazy assets (WP Rocket data-lazy-src, IntersectionObserver,
  63  |       // content-visibility overrides). Ensures carousel JS has initialised
  64  |       // before the element scan runs.
  65  |       await test.step('Preparing Page (Load All Assets)', async () => {
  66  |         await preparePage();
  67  |       });
  68  | 
  69  |       // ── Phase 2: Interaction Audit ─────────────────────────────────
  70  |       let auditResult;
  71  |       await test.step('Running Interaction Audit', async () => {
  72  |         auditResult = await runInteractionAudit(page, endpoint.id);
  73  |       });
  74  | 
  75  |       const { allResults, counts, fails, warns } = auditResult;
  76  | 
  77  |       // ── Annotations (visible in HTML report) ──────────────────────
  78  |       test.info().annotations.push({
  79  |         type: 'Interaction Summary',
  80  |         description: `PASS:${counts.PASS} FAIL:${counts.FAIL} WARN:${counts.WARN} SKIP:${counts.SKIP} | Project:${projectName}`,
  81  |       });
  82  | 
  83  |       if (warns.length) {
  84  |         test.info().annotations.push({
  85  |           type: 'Warnings',
  86  |           description: warns.map(w => `[${w.category}] ${w.label}: ${w.detail}`).join(' | '),
  87  |         });
  88  |       }
  89  |
